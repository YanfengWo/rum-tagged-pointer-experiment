# Golden file: scoping.txt
# Format:  input ||| expected_output
# With
With[{x = 5}, x^2] ||| 25
With[{x = 2, y = 3}, x + y] ||| 5
# Ham limitation: does not simplify (a+b)*(a+b) to (a+b)^2
# With[{x = a + b}, x * x] ||| (a + b)^2
# Module (sequential init)
Module[{x = 1, y = 2}, x + y] ||| 3
# Function definitions and lookup
f[x_] := x^2; f[3] ||| 9
f[x_] := x + 1; f[f[5]] ||| 7
# Nest, NestList, FixedPoint
Nest[# + 1 &, 0, 5] ||| 5
NestList[# * 2 &, 1, 4] ||| {1, 2, 4, 8, 16}
FixedPoint[Floor[#/2] &, 100] ||| 0
# Pure function variations
Function[x, x^2][4] ||| 16
Function[{x, y}, x + y][3, 5] ||| 8
